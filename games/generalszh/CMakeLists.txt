cmake_minimum_required(VERSION 4.4.3)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(generalszh_gameplay LANGUAGES CXX)
	set(CMAKE_CXX_STANDARD 23)
	set(CMAKE_CXX_STANDARD_REQUIRED ON)
	set(CMAKE_CXX_EXTENSIONS OFF)
	include(CTest)
	get_filename_component(generalszh_repo_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
	set(RTS_BUILD_ZEROHOUR_EXTRAS ${BUILD_TESTING})
	# Asset-free modern game/rule/adapter validation, using the same pinned
	# toolchain validation as the engine. No legacy renderer or platform build.
	add_subdirectory("${generalszh_repo_root}/engine" "${CMAKE_CURRENT_BINARY_DIR}/engine")
endif()
